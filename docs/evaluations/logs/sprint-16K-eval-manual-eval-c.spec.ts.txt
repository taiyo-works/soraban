import fs from "node:fs";
import { expect, request as pwRequest, test } from "@playwright/test";
import { measureTextZoom } from "../field-checks";
const BASE = "http://127.0.0.1:3400";
const OUT = "/home/user/soraban/docs/evaluations/logs/sprint-16K-eval-manual-results.jsonl";
const rec = (o: Record<string, unknown>) => fs.appendFileSync(OUT, JSON.stringify(o) + "\n");
const DAY = "2027-01-05";
const at = (hms: string) => new Date(`${DAY}T${hms}+09:00`);
test.use({ serviceWorkers: "allow", hasTouch: true, viewport: { width: 390, height: 844 } });
test("textzoom field", async ({ page, context }) => {
  const m = await pwRequest.newContext({ baseURL: BASE });
  const si = { data: { authUserId: "a-manager", email: "a-manager@example.test", emailVerified: true, name: "a-manager" } };
  await m.post("/api/test-auth/sign-in", si); await page.request.post("/api/test-auth/sign-in", si);
  const mark = `JUZ${Date.now().toString(36).toUpperCase()}`.slice(0, 20);
  const a = (await (await m.post("/api/ops/aircraft", { data: { registrationMark: mark, modelName: "評価機", manufacturer: "DJI", serialNumber: `SN-${mark}`, typeCertified: false, initialFlightMinutes: 0, maxTakeoffWeightKg: 0.9 } })).json()) as { id: string };
  const p = (await (await m.post("/api/ops/pilots", { data: { fullName: `評価 Z`, hasLicense: false } })).json()) as { id: string };
  const pl = (await (await m.post("/api/ops/plans", { data: { title: `Z`, aircraftId: a.id, pilotId: p.id, plannedStartAt: `${DAY}T13:00`, durationMinutes: 60, purpose: "x", placeName: "y", latitude: 35.434238, longitude: 133.380508, flags: [] } })).json()) as { id: string };
  await page.clock.setFixedTime(at("11:56:00"));
  await page.goto(`/ops/plans/${pl.id}/field`);
  await expect(page.locator('[data-testid="field-root"][data-hydrated="true"]')).toBeVisible();
  await page.waitForFunction(() => navigator.serviceWorker.controller !== null, null, { timeout: 60_000 });
  await page.waitForTimeout(2500);
  for (const w of [390, 430]) {
    const r = await measureTextZoom(page, w);
    rec({ kind: "textzoom", state: "field-fresh-online", w, sw: r.scrollWidth, cw: r.clientWidth, clipped: r.clipped, clippedContainers: r.clippedContainers, unscaled: r.unscaled.slice(0, 5), unreachablePrimary: r.unreachablePrimary, navCoversLast: r.navCoversLast });
  }
  await context.setOffline(true);
  await page.evaluate(() => window.dispatchEvent(new Event("offline")));
  await page.clock.setFixedTime(new Date(at("12:00:00").getTime() + 13 * 3600_000));
  await page.reload();
  await expect(page.locator('[data-testid="field-root"][data-hydrated="true"]')).toBeVisible();
  for (const w of [390, 430]) {
    const r = await measureTextZoom(page, w);
    rec({ kind: "textzoom", state: "field-stale-offline", w, sw: r.scrollWidth, cw: r.clientWidth, clipped: r.clipped, clippedContainers: r.clippedContainers, unscaled: r.unscaled.slice(0, 5), unreachablePrimary: r.unreachablePrimary, navCoversLast: r.navCoversLast });
  }
});
