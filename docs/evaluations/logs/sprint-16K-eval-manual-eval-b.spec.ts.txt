import fs from "node:fs";
import { expect, request as pwRequest, test } from "@playwright/test";
const BASE = "http://127.0.0.1:3400";
const OUT = "/home/user/soraban/docs/evaluations/logs/sprint-16K-eval-manual-results.jsonl";
const rec = (o: Record<string, unknown>) => fs.appendFileSync(OUT, JSON.stringify(o) + "\n");
const DAY = "2027-01-05";
const at = (hms: string) => new Date(`${DAY}T${hms}+09:00`);
test.use({ serviceWorkers: "allow", hasTouch: true });
for (const w of [320, 375, 390]) {
  test(`overflow200 ${w}`, async ({ page, context }) => {
    await page.setViewportSize({ width: w, height: 740 });
    const m = await pwRequest.newContext({ baseURL: BASE });
    await m.post("/api/test-auth/sign-in", { data: { authUserId: "a-manager", email: "a-manager@example.test", emailVerified: true, name: "a-manager" } });
    await page.request.post("/api/test-auth/sign-in", { data: { authUserId: "a-manager", email: "a-manager@example.test", emailVerified: true, name: "a-manager" } });
    const mark = `JUO${Date.now().toString(36).toUpperCase()}${w}`.slice(0, 20);
    const a = (await (await m.post("/api/ops/aircraft", { data: { registrationMark: mark, modelName: "評価機", manufacturer: "DJI", serialNumber: `SN-${mark}`, typeCertified: false, initialFlightMinutes: 0, maxTakeoffWeightKg: 0.9 } })).json()) as { id: string };
    const p = (await (await m.post("/api/ops/pilots", { data: { fullName: `評価 ${w}`, hasLicense: false } })).json()) as { id: string };
    const pl = (await (await m.post("/api/ops/plans", { data: { title: `O ${w}`, aircraftId: a.id, pilotId: p.id, plannedStartAt: `${DAY}T13:00`, durationMinutes: 60, purpose: "x", placeName: "y", latitude: 35.434238, longitude: 133.380508, flags: [] } })).json()) as { id: string };
    await page.clock.setFixedTime(at("11:56:00"));
    await page.goto(`/ops/plans/${pl.id}/field`);
    await expect(page.locator('[data-testid="field-root"][data-hydrated="true"]')).toBeVisible();
    await page.waitForFunction(() => navigator.serviceWorker.controller !== null, null, { timeout: 60_000 });
    await page.waitForTimeout(2500);
    const run = async (label: string) => {
      const r = await page.evaluate(() => {
        const cw = document.documentElement.clientWidth;
        const over: string[] = [];
        for (const e of Array.from(document.querySelectorAll("body *"))) {
          const b = e.getBoundingClientRect();
          if (b.width > 0 && b.right > cw + 1) over.push(`${e.getAttribute("data-testid") ?? e.tagName}:${Math.round(b.right)}`);
        }
        return { sw: document.documentElement.scrollWidth, cw, over: over.slice(0, 8), bandH: document.querySelector('[data-testid="field-status-band"]')?.getBoundingClientRect().height };
      });
      rec({ kind: "ov200", w, label, ...r });
    };
    await context.setOffline(true);
    await page.evaluate(() => window.dispatchEvent(new Event("offline")));
    await page.clock.setFixedTime(new Date(at("12:00:00").getTime() + 13 * 3600_000));
    await page.reload();
    await expect(page.locator('[data-testid="field-root"][data-hydrated="true"]')).toBeVisible();
    await page.addStyleTag({ content: "html{font-size:200% !important}" });
    await page.waitForTimeout(300);
    await run("stale200");
    await page.addStyleTag({ content: ".whitespace-nowrap{white-space:normal !important}" });
    await page.waitForTimeout(300);
    await run("stale200-nowrap-off");
    await page.screenshot({ path: `/home/user/soraban/docs/evaluations/screenshots/sprint-16K-eval/${w}-6c-200-nowrap-off.png` });
    // 通常（古くない）の 200%
    await context.setOffline(false);
    await page.clock.setFixedTime(at("11:57:00"));
    await page.reload({ waitUntil: "domcontentloaded" });
    await expect(page.locator('[data-testid="field-root"][data-hydrated="true"]')).toBeVisible();
    await page.addStyleTag({ content: "html{font-size:200% !important}" });
    await page.waitForTimeout(300);
    await run("fresh200-online");
  });
}
