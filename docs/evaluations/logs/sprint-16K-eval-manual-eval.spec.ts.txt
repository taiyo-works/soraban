/** 見えている主ボタン（data-variant）と、主色（#1F5FE0）の面を持つ見えている要素の数 */
const primaryProbe = async (page: Page, root: string) =>
  page.evaluate((sel) => {
    const rootEl = document.querySelector(sel) ?? document.body;
    const vis = (e: Element) => {
      const r = e.getBoundingClientRect();
      return r.width > 0 && r.height > 0 && getComputedStyle(e).visibility !== "hidden";
    };
    const ids = Array.from(rootEl.querySelectorAll('[data-variant="primary"]')).filter(vis).map((e) => e.getAttribute("data-testid") ?? e.tagName);
    const blue = Array.from(rootEl.querySelectorAll("button, a, [role=radio]"))
      .filter(vis)
      .filter((e) => getComputedStyle(e).backgroundColor === "rgb(31, 95, 224)")
      .map((e) => e.getAttribute("data-testid") ?? e.tagName);
    return { ids, blue };
  }, root);
const sunlight = async (page: Page, root: string, min: number) => {
  const G = 0.1;
  const samples = (await collectTextSamples(page, root)).filter((s) => !s.disabled);
  const worst = samples
    .map((s) => {
      const a = relativeLuminance(s.fg);
      const b = relativeLuminance(s.bg);
      return { id: s.testId ?? s.tag, r: (Math.max(a, b) + 0.05 + G) / (Math.min(a, b) + 0.05 + G) };
    })
    .sort((x, y) => x.r - y.r)[0];
  return { n: samples.length, worst, ok: worst ? worst.r >= min : false };
};
const standards = async (page: Page, root: string, label: string, big?: string) => {
  await waitForSettledStyles(page);
  const out: Record<string, unknown> = { kind: "standards", label };
  for (const [name, fn] of [
    ["tap", () => expectTapTargets(page, root, label, big ? { bigTestId: big } : {})],
    ["font18", () => expectFontSize(page, root, 18, label)],
    ["contrast7", () => expectContrast(page, root, 7, label)],
    ["noHScroll", () => expectNoHorizontalScroll(page, label)],
  ] as const) {
    try {
      await fn();
      out[name] = "ok";
    } catch (e) {
      out[name] = "FAIL " + String(e).slice(0, 300);
    }
  }
  out.sun = await sunlight(page, root, 4.5);
  out.mode = await page.locator("html").getAttribute("data-mode");
  out.viewport = page.viewportSize();
  rec(out);
};
const dumpStorage = async (page: Page) =>
  page.evaluate(async () => {
    const ls: Record<string, string> = {};
    for (let i = 0; i < localStorage.length; i++) {
      const k = localStorage.key(i)!;
      ls[k] = localStorage.getItem(k) ?? "";
    }
    const ss: Record<string, string> = {};
    for (let i = 0; i < sessionStorage.length; i++) {
      const k = sessionStorage.key(i)!;
      ss[k] = sessionStorage.getItem(k) ?? "";
    }
    const idb: Record<string, Record<string, string[]>> = {};
    const idbValues: string[] = [];
    const dbs = (await indexedDB.databases?.()) ?? [];
    for (const d of dbs) {
      if (!d.name) continue;
      await new Promise<void>((res) => {
        const r = indexedDB.open(d.name!);
        r.onsuccess = async () => {
          const db = r.result;
          idb[d.name!] = {};
          for (const s of Array.from(db.objectStoreNames)) {
            await new Promise<void>((res2) => {
              const tx = db.transaction(s, "readonly");
              const st = tx.objectStore(s);
              const kq = st.getAllKeys();
              const vq = st.getAll();
              tx.oncomplete = () => {
                idb[d.name!]![s] = kq.result.map(String);
                for (const v of vq.result) idbValues.push(`${s}::` + JSON.stringify(v));
                res2();
              };
              tx.onerror = () => res2();
            });
          }
          db.close();
          res();
        };
        r.onerror = () => res();
      });
    }
    const cacheMap: Record<string, string[]> = {};
    for (const n of await self.caches.keys()) cacheMap[n] = (await (await self.caches.open(n)).keys()).map((q) => new URL(q.url).pathname);
    return { ls, ss, idb, idbValues, caches: cacheMap };
  });

    rec({ kind: "forbidden-in-storage-while-pending", pilotName: blob.includes(PILOT_NAME), mark: blob.includes(registrationMark), lat: blob.includes("35.4342"), lng: blob.includes("133.3805"), note: blob.includes(NOTE) });
    // 通信を戻す
    await setOnline(page, context, true);
    await expect.poll(async () => page.getByTestId("field-status-unsent").getAttribute("data-count"), { timeout: 60_000 }).toBe("0");
    rec({ kind: "flushed", unsent: await page.getByTestId("field-status-unsent").getAttribute("data-count") });
    await page.reload();
    await expect(page.locator('[data-testid="field-root"][data-hydrated="true"]')).toBeVisible();
    const rows = await page.getByTestId("field-flight-row").count();
    const phases = await page.getByTestId("field-flight-row").evaluateAll((els) => els.map((e) => e.getAttribute("data-phase")));
    rec({ kind: "after-flush-rows", rows, phases, journalLinks: await page.getByTestId("field-journal-link").count() });
    // サーバー側の件数
    const fl = await manager.get(`/api/ops/flights`);
    const body = fl.ok() ? ((await fl.json()) as unknown) : null;
    const list = Array.isArray(body) ? body : ((body as { flights?: unknown[]; items?: unknown[] } | null)?.flights ?? (body as { items?: unknown[] } | null)?.items ?? []);
    const mine = (list as Array<{ aircraftRegistrationMark?: string; registrationMark?: string; aircraft?: { registrationMark?: string }; safetyIssue?: boolean }>).filter((f) => JSON.stringify(f).includes(registrationMark));
    rec({ kind: "server-flights", status: fl.status(), total: (list as unknown[]).length, mine: mine.length });
    const after = await dumpStorage(page);
    const blob2 = JSON.stringify(after);
    rec({ kind: "storage-after", ls: Object.keys(after.ls), idb: after.idb, caches: after.caches, pilotName: blob2.includes(PILOT_NAME), mark: blob2.includes(registrationMark), lat: blob2.includes("35.4342"), note: blob2.includes(NOTE) });
    rec({ kind: "pageErrors", errors: pageErrors });
    await manager.dispose();
  });
});
